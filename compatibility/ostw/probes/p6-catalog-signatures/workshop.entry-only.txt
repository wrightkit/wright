variables
{
    global:
        0: hudText
        1: toggle
        2: counter
}

// Rule Element Count: 6
rule("Initial Global")
{
    event
    {
        Ongoing - Global;
    }

    // Element Count: 5, Action Count: 3
    // Largest Action Index: 0 using 2 Elements
    actions
    {
        Set Global Variable(hudText, 0);
        Set Global Variable(toggle, False);
        Set Global Variable(counter, 0);
    }
}

// Rule Element Count: 62
rule("P6 named arguments against catalog signatures")
{
    event
    {
        Ongoing - Each Player;
        All;
        All;
    }

    // Element Count: 61, Action Count: 13
    // Largest Action Index: 10 using 12 Elements
    actions
    {
        Set Global Variable(toggle, Workshop Setting Toggle(Custom String("6. Interface"), Custom String("Highlight hovered hero (4 texts)"), True, 5));
        Create In-World Text(All Players(All Teams), Custom String("{0}", Global Variable(toggle)), Subtract(Left, Left), 3, Do Not Clip, Visible To And String, Color(White), Visible Always);
        Set Global Variable(hudText, Last Text ID);
        Big Message(All Players(All Teams), Custom String("{0}", Global Variable(toggle)));
        Start Camera(Event Player, Subtract(Left, Left), Left, 0);
        Modify Global Variable(counter, Add, 12);
        If(Is Button Held(Event Player, Button(Ability 2)));
            Wait(0.05, Abort When False);
        End;
        Set Invisible(Event Player, All);
        Create HUD Text(If-Then-Else(Global Variable(toggle), All Players(All Teams), Null), Custom String("banned"), Null, Null, Left, 1, Color(Red), Color(White), Color(White), Visible To, Visible Always);
        Small Message(Event Player, Custom String("p6 done"));
        Big Message(All Players(All Teams), Custom String("{0}|{1}", Global Variable(hudText), Global Variable(counter)));
    }
}
