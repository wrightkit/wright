variables
{
    global:
        0: favorite
        1: combo
        2: selector
        3: counter
}

// Rule Element Count: 3
rule("Initial Global")
{
    event
    {
        Ongoing - Global;
    }

    // Element Count: 2, Action Count: 1
    actions
    {
        Set Global Variable(counter, 0);
    }
}

// Rule Element Count: 44
rule("P4 types: enum, union, null, cast")
{
    event
    {
        Ongoing - Each Player;
        All;
        All;
    }

    // Element Count: 43, Action Count: 12
    // Largest Action Index: 10 using 11 Elements
    actions
    {
        Set Global Variable(favorite, 1);
        Set Global Variable(combo, Null);
        If(Compare(Global Variable(combo), ==, Null));
            Big Message(Event Player, Custom String("combo is null"));
        End;
        Set Global Variable(combo, Team 2);
        If(Compare(Global Variable(combo), ==, Team 1));
            Big Message(Event Player, Custom String("team one"));
        End;
        Set Global Variable(selector, Global Variable(combo));
        Set Global Variable(selector, If-Then-Else(Compare(Global Variable(selector), >, 0), 10, 20));
        Big Message(Event Player, Custom String("{0}|{1}|{2}", Global Variable(favorite), Global Variable(combo), Global Variable(selector)));
    }
}

// Rule Element Count: 48
rule("P4 short circuit pure operands")
{
    event
    {
        Ongoing - Each Player;
        All;
        All;
    }

    // Element Count: 47, Action Count: 8
    // Largest Action Index: 0 using 13 Elements
    actions
    {
        If(And(Compare(Global Variable(selector), >, 0), Compare(Global Variable(selector), <, 100)));
            Big Message(Event Player, Custom String("range ok"));
        End;
        If(Or(Compare(Global Variable(selector), ==, 0), Compare(Global Variable(selector), ==, 1)));
            Big Message(Event Player, Custom String("edge"));
        End;
        Set Global Variable(selector, If-Then-Else(Compare(Global Variable(selector), >, 0), 10, 20));
        Big Message(Event Player, Custom String("{0}", Global Variable(selector)));
    }
}

// Rule Element Count: 32
rule("P4 short circuit actionful operands")
{
    event
    {
        Ongoing - Each Player;
        All;
        All;
    }

    // Element Count: 31, Action Count: 9
    // Largest Action Index: 7 using 11 Elements
    actions
    {
        Modify Global Variable(counter, Add, 1);
        Modify Global Variable(counter, Add, 1);
        If(True);
            Big Message(Event Player, Custom String("both pings"));
        End;
        Modify Global Variable(counter, Add, 1);
        Set Global Variable(selector, If-Then-Else(True, 10, 20));
        Set Global Variable(selector, If-Then-Else(Compare(Global Variable(selector), >, 0), 1, 2));
        Big Message(Event Player, Custom String("{0}", Global Variable(selector)));
    }
}
