variables
{
    global:
        0: counter
    player:
        0: by
        1: by_0
        2: by_1
        3: by_2
        4: k
        5: by_3
        6: by_4
        7: by_5
        8: by_6
}

// Rule Element Count: 3
rule("Initial Global")
{
    event
    {
        Ongoing - Global;
    }

    // Element Count: 2, Action Count: 1
    actions
    {
        Set Global Variable(counter, 0);
    }
}

// Rule Element Count: 117
rule("P5 functions, control flow, switch, return")
{
    event
    {
        Ongoing - Each Player;
        All;
        All;
    }

    // Element Count: 116, Action Count: 30
    // Largest Action Index: 18 using 21 Elements
    actions
    {
        Set Global Variable(counter, 6);
        Set Player Variable(Event Player, by, 4);
        Modify Global Variable(counter, Add, Player Variable(Event Player, by));
        Set Player Variable(Event Player, by_0, 5);
        Modify Global Variable(counter, Add, Player Variable(Event Player, by_0));
        Set Player Variable(Event Player, by_1, 1);
        Modify Global Variable(counter, Add, Player Variable(Event Player, by_1));
        If(Compare(Global Variable(counter), ==, 0));
            Abort;
        End;
        For Global Variable(counter, 0, Compare(Global Variable(counter), <, 5), 1);
            Set Player Variable(Event Player, by_2, 1);
            Modify Global Variable(counter, Add, Player Variable(Event Player, by_2));
            Wait(0.1, Ignore Condition);
        End;
        For Player Variable(Event Player, k, 0, Count Of(Array(1, 2, 3)), 1);
            Modify Global Variable(counter, Add, Value In Array(Array(1, 2, 3), Player Variable(Event Player, k)));
        End;
        Skip(Value In Array(Array(7, 0, 2, 5), Add(Index Of Array Value(Array(1, 2, 3), Global Variable(counter)), 1)));
        Set Player Variable(Event Player, by_3, 10);
        Modify Global Variable(counter, Add, Player Variable(Event Player, by_3));
        Set Player Variable(Event Player, by_4, 20);
        Modify Global Variable(counter, Add, Player Variable(Event Player, by_4));
        Skip(4);
        Set Player Variable(Event Player, by_5, 30);
        Modify Global Variable(counter, Add, Player Variable(Event Player, by_5));
        Set Player Variable(Event Player, by_6, 40);
        Modify Global Variable(counter, Add, Player Variable(Event Player, by_6));
        Set Global Variable(counter, Multiply(Global Variable(counter), 2));
        Big Message(Event Player, Custom String("{0}", Global Variable(counter)));
    }
}
